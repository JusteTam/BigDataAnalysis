#Indexing

hisat2-build -p 6 ~/HW1/references/GRCh38.primary_assembly.genome.fa indexed_reference_2

#QC
fastqc *.fastq.gz -o ~/HW2/raw_data/fastqc_results
fastqc SRR11647663_1.fastq.gz -o ~/HW2/raw_data/fastqc_results

#Trimming
trim_galore -j 6 --paired --quality 20 --gzip --fastqc SRR11647652_1.fastq.gz SRR11647652_2.fastq.gz -o ~/HW2/trimming_results
trim_galore -j 6 --paired --quality 20 --gzip --fastqc SRR11647653_1.fastq.gz SRR11647653_2.fastq.gz -o ~/HW2/trimming_results
trim_galore -j 6 --paired --quality 20 --gzip --fastqc SRR11647662_1.fastq.gz SRR11647662_2.fastq.gz -o ~/HW2/trimming_results
trim_galore -j 6 --paired --quality 20 --gzip --fastqc SRR11647663_1.fastq.gz SRR11647663_2.fastq.gz -o ~/HW2/trimming_results


#Mapping komanda keiciam i bsmap - teams bus komanda
conda create -n bsmap_env bioconda::bsmap
conda activate bsmap_env
bsmap -a ~/HW2/trimming_results/SRR11647652_1_val_1.fq.gz -b ~/HW2/trimming_results/SRR11647652_2_val_2.fq.gz -d ~/HW1/references/GRCh38.primary_assembly.genome.fa -o SRR11647652.bam -p 6
bsmap -a ~/HW2/trimming_results/SRR11647653_1_val_1.fq.gz -b ~/HW2/trimming_results/SRR11647653_2_val_2.fq.gz -d ~/HW1/references/GRCh38.primary_assembly.genome.fa -o SRR11647653.bam -p 6
bsmap -a ~/HW2/trimming_results/SRR11647662_1_val_1.fq.gz -b ~/HW2/trimming_results/SRR11647662_2_val_2.fq.gz -d ~/HW1/references/GRCh38.primary_assembly.genome.fa -o SRR11647662.bam -p 6
bsmap -a ~/HW2/trimming_results/SRR11647663_1_val_1.fq.gz -b ~/HW2/trimming_results/SRR11647663_2_val_2.fq.gz -d ~/HW1/references/GRCh38.primary_assembly.genome.fa -o SRR11647663.bam -p 6

#BAM failu sortinimas

picard AddOrReplaceReadGroups I=SRR11647653.sorted.bam O=SRR11647653.rg.bam RGID=SRR11647653 RGLB=lib1 RGPL=ILLUMINA RGPU=unit1 RGSM=SRR11647653
picard AddOrReplaceReadGroups I=SRR11647662.sorted.bam O=SRR11647662.rg.bam RGID=SRR11647662 RGLB=lib1 RGPL=ILLUMINA RGPU=unit1 RGSM=SRR11647662
picard AddOrReplaceReadGroups I=SRR11647663.sorted.bam O=SRR11647663.rg.bam RGID=SRR11647663 RGLB=lib1 RGPL=ILLUMINA RGPU=unit1 RGSM=SRR11647663
